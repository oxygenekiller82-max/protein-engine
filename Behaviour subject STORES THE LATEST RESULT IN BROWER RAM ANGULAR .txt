import { BehaviorSubject } from 'rxjs';

@Injectable({ providedIn: 'root' })
export class SequenceService {
  // 1. Create the "Private Storage" in RAM
  private historyData = new BehaviorSubject<any>(null);

  // 2. Create the "Public Stream" for components to listen to
  currentHistory$ = this.historyData.asObservable();

  constructor(private http: HttpClient) {}

  generatePeptide(params: any) {
    return this.http.post<any>('your-api-url', params).pipe(
      tap(response => {
        // 3. When data arrives, save it to the "Private Storage"
        this.historyData.next(response);
      })
    );
  }
}
